# 函数式编程范式

*类型：knowledge ｜ 难度：进阶 ｜ 标签：函数式编程、柯里化、函数组合、偏函数、记忆化、惰性求值*

**结论：函数式编程的核心思路是「用纯函数 + 函数组合代替过程式的状态修改」：柯里化把多参函数拆成单参链，实现参数复用与延迟执行；函数组合（compose/pipe）把小函数拼成数据处理管道；偏函数固定部分参数生成特化函数；记忆化用缓存换时间，只对纯函数有效；惰性求值把计算推迟到真正需要时。这些技术的前提都是纯函数——一旦有副作用，缓存与组合都会失效。**

## 函数柯里化（Currying）

柯里化是把接受多个参数的函数转换成一系列接受单一参数的函数的技术：每次传入一部分参数，参数收齐才真正执行。

```js
// 普通函数：一次传齐所有参数
function add(a, b, c) {
  return a + b + c;
}
console.log(add(1, 2, 3)); // 6

// 手工柯里化：每层只收一个参数，收齐才计算
function curriedAdd(a) {
  return function(b) {
    return function(c) {
      return a + b + c;
    };
  };
}
console.log(curriedAdd(1)(2)(3)); // 6

// ES6 箭头函数写法
const curriedAdd2 = a => b => c => a + b + c;
```

### 通用柯里化函数

```js
function curry(fn) {
  return function curried(...args) {
    // 实参数量达到形参声明数量（fn.length），直接执行
    if (args.length >= fn.length) {
      return fn.apply(this, args);
    }
    // 参数不够：返回新函数继续收集，已收参数通过闭包保存
    return function(...moreArgs) {
      return curried.apply(this, args.concat(moreArgs));
    };
  };
}

const curriedSum = curry(function(a, b, c, d) {
  return a + b + c + d;
});

// 支持任意拆分方式传参
console.log(curriedSum(1)(2)(3)(4)); // 10
console.log(curriedSum(1, 2)(3, 4)); // 10
console.log(curriedSum(1)(2, 3, 4)); // 10
```

### 柯里化的三个收益

参数复用：

```js
const multiply = curry((a, b) => a * b);
const double = multiply(2); // 固定第一个参数，得到特化函数
const triple = multiply(3);
console.log(double(5)); // 10
console.log(triple(5)); // 15

// 实际应用：日志函数固定级别与时间，只留消息参数
const log = curry((level, time, message) => {
  console.log(`[${level}] ${time}: ${message}`);
});
const errorLogNow = log('ERROR')(new Date().toISOString());
errorLogNow('Database connection failed');
errorLogNow('API timeout');
```

延迟执行：

```js
const calculate = curry((operation, a, b) => {
  switch (operation) {
    case 'add': return a + b;
    case 'subtract': return a - b;
    case 'multiply': return a * b;
    case 'divide': return a / b;
  }
});

// 先配置操作类型，拿到待命函数；真正需要时再传数值
const add = calculate('add');
const multiply = calculate('multiply');
console.log(add(5, 3));      // 8
console.log(multiply(5, 3)); // 15
```

提前确认：

```js
const match = curry((regex, str) => regex.test(str));

// 先固化校验规则，之后反复使用
const hasNumber = match(/[0-9]+/);
const hasLetter = match(/[a-zA-Z]+/);
console.log(hasNumber('abc123')); // true
console.log(hasLetter('123'));    // false
```

## 函数组合（Function Composition）

函数组合把多个单参函数串成一个管道：前一个函数的输出作为后一个函数的输入。

```js
// compose：从右往左执行
const compose = (...fns) => x => fns.reduceRight((acc, fn) => fn(acc), x);

// pipe：从左往右执行，阅读顺序与执行顺序一致
const pipe = (...fns) => x => fns.reduce((acc, fn) => fn(acc), x);

const addOne = x => x + 1;
const double = x => x * 2;
const square = x => x * x;

// 两条管道等价：((3^2) * 2) + 1 = 19
console.log(compose(addOne, double, square)(3)); // 19
console.log(pipe(square, double, addOne)(3));    // 19

// 实际应用：数据处理管道
const users = [
  { name: 'Alice', age: 25, active: true },
  { name: 'Bob', age: 30, active: false },
  { name: 'Charlie', age: 35, active: true }
];

const getActiveUsers = list => list.filter(u => u.active);
const getUserNames = list => list.map(u => u.name);
const sortNames = names => names.sort();

const processUsers = pipe(getActiveUsers, getUserNames, sortNames);
console.log(processUsers(users)); // ['Alice', 'Charlie']
```

组合的前提是每个环节都是单参纯函数；中间任何一环有副作用，管道的推理与复用都会被破坏。

## 偏函数（Partial Application）

偏函数固定函数的部分参数，返回一个接受剩余参数的新函数。与柯里化的区别：柯里化每次只收一个参数，偏函数可以一次固定多个。

```js
function partial(fn, ...presetArgs) {
  return function(...laterArgs) {
    // 预置参数在前，调用时参数在后
    return fn(...presetArgs, ...laterArgs);
  };
}

function greet(greeting, name, punctuation) {
  return `${greeting}, ${name}${punctuation}`;
}

const sayHelloTo = partial(greet, 'Hello');
console.log(sayHelloTo('Alice', '!')); // "Hello, Alice!"
console.log(sayHelloTo('Bob', '.'));   // "Hello, Bob."

const sayHelloToAlice = partial(greet, 'Hello', 'Alice');
console.log(sayHelloToAlice('!')); // "Hello, Alice!"
```

```js
// 柯里化：严格的单参链
const curriedAdd = a => b => c => a + b + c;
curriedAdd(1)(2)(3); // 6

// 偏函数：一次固定任意多个参数，更直接
const partialAdd = partial((a, b, c) => a + b + c, 1);
partialAdd(2, 3); // 6
```

## 记忆化（Memoization）

记忆化通过缓存函数的计算结果来避免重复计算，只对纯函数有效——输入相同必然输出相同，缓存才安全。

```js
function memoize(fn) {
  const cache = new Map();

  return function(...args) {
    // 以参数序列化结果作为缓存键（仅适用于可序列化参数）
    const key = JSON.stringify(args);

    if (cache.has(key)) {
      return cache.get(key); // 命中缓存，不再计算
    }

    const result = fn.apply(this, args);
    cache.set(key, result);
    return result;
  };
}

function fibonacci(n) {
  if (n <= 1) return n;
  return fibonacci(n - 1) + fibonacci(n - 2);
}

const memoizedFib = memoize(fibonacci);
console.log(memoizedFib(10)); // 55（首次计算）
console.log(memoizedFib(10)); // 55（直接读缓存）
```

工程细节：`JSON.stringify` 做键无法区分 `undefined` 与缺失参数等边界；长期存活的缓存会持有参数与结果的引用，必要时配合 LRU 策略或在键为对象时使用 `WeakMap` 控制内存。

## 惰性求值（Lazy Evaluation）

惰性求值把计算推迟到第一次真正需要结果时，之后的读取直接复用。

```js
function createLazyValue(fn) {
  let cached = false;
  let value;

  return function() {
    if (!cached) { // 第一次调用才执行 fn
      value = fn();
      cached = true;
    }
    return value; // 之后全部复用
  };
}

const lazyExpensiveCalc = createLazyValue(() => {
  console.log('Computing expensive value...');
  return 42 * 1000;
});

console.log('Lazy value created');  // 创建时不计算
console.log(lazyExpensiveCalc());   // Computing expensive value... 42000
console.log(lazyExpensiveCalc());   // 42000（不再计算）
```

```js
// 实际应用：连接等昂贵资源延迟到首次使用时初始化
function createConnection() {
  let connection = null;

  return {
    getConnection() {
      if (!connection) {
        console.log('Establishing connection...');
        connection = { status: 'connected' };
      }
      return connection;
    }
  };
}

const db = createConnection();
console.log(db.getConnection()); // Establishing connection... { status: 'connected' }
console.log(db.getConnection()); // 直接返回已建立的连接
```

## 实践要点

- 用纯函数作为最小单元，再通过组合搭建流程，副作用集中推到管道边界
- 柯里化与偏函数适合「配置先行、数据后到」的场景
- 记忆化只用于纯函数，缓存键的生成要考虑参数类型与内存占用
- 避免为了函数式而函数式：嵌套过深的组合比直白的循环更难读
