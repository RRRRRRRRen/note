# 高阶函数

*类型：knowledge ｜ 难度：进阶 ｜ 标签：高阶函数、map、filter、reduce、once、debounce、throttle、组合*

**结论：高阶函数是「以函数为输入或输出」的函数——接受函数作参数，或返回一个函数（至少满足其一）。它把「行为」变成可以传递和组装的值，是 JavaScript 一等函数能力的直接产物：map/filter/reduce 用回调抽象「遍历模式」，once/debounce/throttle 用闭包包裹原函数实现执行控制，pipe/compose 用函数拼装数据管道。掌握高阶函数的本质是掌握一种抽象思路：把变化的部分（回调）与不变的部分（控制逻辑）分离。**

## 什么是高阶函数

满足以下至少一个条件的函数即为高阶函数：

- 接受一个或多个函数作为参数
- 返回一个函数作为结果

```js
// 1. 函数作参数：把「做什么」交给调用方
function repeat(n, action) {
  for (let i = 0; i < n; i++) {
    action(i);
  }
}
repeat(3, console.log);
// 0
// 1
// 2

// 2. 函数作返回值：闭包记住 factor，生成特化函数
function multiplier(factor) {
  return function(number) {
    return number * factor;
  };
}

const double = multiplier(2);
const triple = multiplier(3);
console.log(double(5)); // 10
console.log(triple(5)); // 15
```

## 数组高阶函数与手动实现

### map

对每个元素应用回调，返回由返回值组成的新数组。

```js
const numbers = [1, 2, 3, 4, 5];
const doubled = numbers.map(n => n * 2);
console.log(doubled); // [2, 4, 6, 8, 10]

// 手动实现：核心是「收集每次回调的返回值」
Array.prototype.myMap = function(callback) {
  const result = [];
  for (let i = 0; i < this.length; i++) {
    result.push(callback(this[i], i, this));
  }
  return result;
};
```

### filter

返回使回调结果为真的元素组成的新数组。

```js
const numbers = [1, 2, 3, 4, 5, 6];
const evens = numbers.filter(n => n % 2 === 0);
console.log(evens); // [2, 4, 6]

// 手动实现：核心是「按回调真值过滤」
Array.prototype.myFilter = function(callback) {
  const result = [];
  for (let i = 0; i < this.length; i++) {
    if (callback(this[i], i, this)) {
      result.push(this[i]);
    }
  }
  return result;
};
```

### reduce

把数组归约为单个值；不传初始值时以第一个元素作为初始累加值。

```js
const numbers = [1, 2, 3, 4, 5];
console.log(numbers.reduce((acc, n) => acc + n, 0));      // 15
console.log(numbers.reduce((acc, n) => Math.max(acc, n))); // 5

// 手动实现：注意无初始值时从第二个元素开始
Array.prototype.myReduce = function(callback, initialValue) {
  let accumulator = initialValue;
  let startIndex = 0;

  if (initialValue === undefined) {
    accumulator = this[0];
    startIndex = 1;
  }

  for (let i = startIndex; i < this.length; i++) {
    accumulator = callback(accumulator, this[i], i, this);
  }
  return accumulator;
};

// 复杂应用：数组转对象（索引映射）
const users = [
  { id: 1, name: 'Alice' },
  { id: 2, name: 'Bob' },
  { id: 3, name: 'Charlie' }
];
const userMap = users.reduce((acc, user) => {
  acc[user.id] = user;
  return acc;
}, {});
// { '1': { id: 1, name: 'Alice' }, ... }
```

### forEach

对每个元素执行回调，无返回值，不能中断循环（`return` 只跳过当次回调）。

```js
const numbers = [1, 2, 3, 4, 5];
numbers.forEach((n, index) => {
  console.log(`Index ${index}: ${n}`);
});

Array.prototype.myForEach = function(callback) {
  for (let i = 0; i < this.length; i++) {
    callback(this[i], i, this);
  }
};
```

### some 与 every

`some` 是「存在量词」：至少一个满足返回真；`every` 是「全称量词」：全部满足才返回真，两者都可短路。

```js
const numbers = [1, 2, 3, 4, 5];
console.log(numbers.some(n => n % 2 === 0)); // true（存在偶数）
console.log(numbers.every(n => n > 0));      // true（全部为正）

Array.prototype.mySome = function(callback) {
  for (let i = 0; i < this.length; i++) {
    if (callback(this[i], i, this)) return true; // 命中即短路
  }
  return false;
};

Array.prototype.myEvery = function(callback) {
  for (let i = 0; i < this.length; i++) {
    if (!callback(this[i], i, this)) return false; // 未命中即短路
  }
  return true;
};
```

## 实用的高阶函数

### once：只执行一次

```js
function once(fn) {
  let called = false;
  let result;

  return function(...args) {
    if (!called) { // 用闭包标记保证只执行一次
      called = true;
      result = fn.apply(this, args);
    }
    return result; // 后续调用返回首次结果
  };
}

const initialize = once(() => {
  console.log('Initializing...');
  return { status: 'initialized' };
});

console.log(initialize()); // Initializing... { status: 'initialized' }
console.log(initialize()); // { status: 'initialized' }（不再执行）
```

### debounce：防抖

连续触发只保留最后一次，适合搜索输入等「结果导向」场景。

```js
function debounce(fn, delay) {
  let timer = null;

  return function(...args) {
    clearTimeout(timer); // 每次触发都作废上一次倒计时
    timer = setTimeout(() => {
      fn.apply(this, args);
    }, delay);
  };
}

const search = debounce((query) => {
  console.log('Searching for:', query);
}, 500);

search('a');
search('ab');
search('abc'); // 只有这一次会执行
```

### throttle：节流

固定时间窗口内至多执行一次，适合滚动等「过程导向」场景。

```js
function throttle(fn, delay) {
  let lastTime = 0;

  return function(...args) {
    const now = Date.now();
    if (now - lastTime >= delay) { // 距上次执行超过窗口才放行
      fn.apply(this, args);
      lastTime = now;
    }
  };
}

window.addEventListener('scroll', throttle(() => {
  console.log('Scroll position:', window.scrollY);
}, 200)); // 无论滚动多快，每 200ms 最多执行一次
```

### pipe 与 compose：函数管道

```js
// pipe：从左往右依次执行
function pipe(...fns) {
  return function(value) {
    return fns.reduce((acc, fn) => fn(acc), value);
  };
}

const processUser = pipe(
  user => ({ ...user, name: user.name.toUpperCase() }),
  user => ({ ...user, age: user.age + 1 }),
  user => ({ ...user, active: true })
);

console.log(processUser({ name: 'alice', age: 25 }));
// { name: 'ALICE', age: 26, active: true }

// compose：从右往左依次执行
function compose(...fns) {
  return function(value) {
    return fns.reduceRight((acc, fn) => fn(acc), value);
  };
}

const addOne = x => x + 1;
const double = x => x * 2;
const square = x => x * x;

const calculate = compose(addOne, double, square);
console.log(calculate(3)); // ((3^2) * 2) + 1 = 19
```

## 实践要点

- 遍历场景优先 map/filter/reduce，替代手写 for 循环中的「模式样板」
- `forEach` 无返回值且不可中断，需要提前退出用 `some`/`every`/`for...of`
- once/debounce/throttle 的通用结构是「闭包保存状态 + 返回包装函数」，理解一个即可举一反三
- 回调函数保持纯函数特性，管道组合才可推理、可复用
